class RepositoryManagementSystem:
    """The repository management system for 50+ repositories."""
    
    def __init__(self):
        self.repositories = {}
        self.synapses = {}
        self.firing_patterns = []
        self.hive_brain = HiveBrain()
        self.biosphere = Biosphere()
    
    def register_repository(self, repo):
        """Register a repository with the Hive."""
        self.repositories[repo.id] = repo
        self.synapses[repo.id] = {}
        self.hive_brain.add_region(repo)
        self.biosphere.add_element(repo)
    
    def connect_repositories(self, source_id, target_id, strength=0.5):
        """Connect two repositories with a synapse."""
        if source_id not in self.repositories or target_id not in self.repositories:
            raise ValueError("Repository not found")
        
        self.synapses[source_id][target_id] = {
            "strength": strength,
            "firings": 0,
            "last_fired": None,
            "type": "bidirectional"
        }
    
    def fire_synapse(self, source_id, target_id, signal):
        """Fire a signal across a synapse."""
        if source_id not in self.synapses or target_id not in self.synapses[source_id]:
            raise ValueError("Synapse not found")
        
        synapse = self.synapses[source_id][target_id]
        synapse["firings"] += 1
        synapse["last_fired"] = datetime.now().isoformat()
        
        # Strengthen the synapse
        synapse["strength"] = min(1.0, synapse["strength"] + 0.01)
        
        self.firing_patterns.append({
            "source": source_id,
            "target": target_id,
            "signal": signal,
            "strength": synapse["strength"],
            "timestamp": datetime.now().isoformat()
        })
    
    def get_repository_status(self, repo_id):
        """Get the status of a repository."""
        if repo_id not in self.repositories:
            return None
        
        repo = self.repositories[repo_id]
        synapses = self.synapses.get(repo_id, {})
        
        return {
            "id": repo_id,
            "name": repo.name,
            "function": repo.function,
            "connections": len(synapses),
            "synaptic_strength": sum(s["strength"] for s in synapses.values()) / len(synapses) if synapses else 0,
            "last_firing": max((s.get("last_fired") for s in synapses.values()), default=None),
            "status": "healthy"
        }
    
    def get_brain_map(self):
        """Get a map of the entire brain."""
        return self.hive_brain.get_map()
    
    def get_biosphere_status(self):
        """Get the status of the biosphere."""
        return self.biosphere.get_status()

class Neurotransmitter:
    """A neurotransmitter model for the Hive."""
    
    def __init__(self, name, function, repositories):
        self.name = name
        self.function = function
        self.repositories = repositories
        self.level = 0.5  # default
        self.release_rate = 0.1
    
    def release(self, amount):
        """Release neurotransmitter."""
        self.level = min(1.0, self.level + amount)
    
    def reuptake(self, amount):
        """Reuptake neurotransmitter."""
        self.level = max(0.0, self.level - amount)
    
    def get_status(self):
        """Get the status of the neurotransmitter."""
        return {
            "name": self.name,
            "function": self.function,
            "repositories": self.repositories,
            "level": self.level,
            "status": "balanced" if 0.3 < self.level < 0.7 else "imbalanced"
        }

class NeuralPathway:
    """A neural pathway for the Hive."""
    
    def __init__(self, source, target, strength, neurotransmitters):
        self.source = source
        self.target = target
        self.strength = strength
        self.neurotransmitters = neurotransmitters
        self.firings = 0
        
    def fire(self, signal):
        """Fire the pathway."""
        self.firings += 1
        # Release neurotransmitters
        for nt in self.neurotransmitters:
            nt.release(0.05)
        return {
            "source": self.source,
            "target": self.target,
            "signal": signal,
            "neurotransmitters": [nt.name for nt in self.neurotransmitters]

class RepositoryHealthMonitor:
    """Monitor the health of all repositories."""
    
    def __init__(self):
        self.health_status = {}
        self.alert_thresholds = {
            "latency": 500,  # milliseconds
            "error_rate": 0.05,  # 5%
            "response_time": 1000,  # milliseconds
            "synaptic_strength": 0.3  # minimum strength
        }
        
    def check_health(self, repo_id, metrics):
        """Check the health of a repository."""
        health = {}
        for metric, value in metrics.items():
            if metric in self.alert_thresholds:
                if value > self.alert_thresholds[metric]:
                    health[metric] = "critical"
                elif value > self.alert_thresholds[metric] * 0.8:
                    health[metric] = "warning"
                else:
                    health[metric] = "healthy"
        self.health_status[repo_id] = health
        return health
    
    def get_system_health(self):
        """Get the overall system health."""
        system_status = "healthy"
        for repo_id, health in self.health_status.items():
            for metric, status in health.items():
                if status == "critical":
                    system_status = "critical"
                    break
        return {"status": system_status, "repositories": self.health_status}
